# Hypercycle dependencies
# Install with: pip install -r requirements.txt
#
# Every entry is a self-contained wheel — no system packages, no compiler, and
# nothing downloaded on first use. That is what keeps the app portable.

pywebview>=5.0

# Raster image encode/decode. AVIF support is native from Pillow 11.3 onward,
# which is why no separate AVIF plugin is listed.
pillow>=11.3

# HEIC/HEIF only. pillow-heif 1.x no longer registers an AVIF opener because
# Pillow handles AVIF itself.
pillow-heif>=1.6.0

# SVG and PDF rasterisation.
PyMuPDF>=1.25.0

# Audio and video conversion. Ships the ffmpeg binary inside the wheel and
# resolves it via get_ffmpeg_exe() — no system ffmpeg, no PATH lookup. Note
# this bundles ffmpeg only, not ffprobe, so duration is read from ffmpeg's own
# stderr rather than a separate probe.
imageio-ffmpeg>=0.6.0

# Document conversion. The _binary distribution bundles the pandoc executable;
# the plain pypandoc package does not. PDF output additionally needs a separate
# PDF engine that pandoc does not bundle, so PDF targets are gated at runtime.
pypandoc_binary>=1.13
